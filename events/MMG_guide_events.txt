# =============================================================================
# MMG_beginners_armaments_guide - menu and detail events
#
# Navigation model used by every chapter:
#   decision menu event -> detail event -> "understood" closes the popup
#                                      -> "I want to see the others" re-fires
#                                         the same chapter menu event.
#
# All events are country-scoped and `is_triggered_only = yes`: they are fired
# by the seven guide decisions or by an event option above them. No country
# state is written, so the events are repeatable and save-safe.
# =============================================================================

add_namespace = MMG_guide
